CREATE TABLE yt_clean (
    video_id INT,
    channel_name TEXT,
    title TEXT,
    category TEXT,
    views BIGINT,
    likes INT,
    dislikes INT,
    comments INT,
    publish_date DATE,
    city TEXT,
    status TEXT
);

desc yt_clean;

INSERT INTO yt_clean (
    video_id, channel_name, title, category,
    views, likes, dislikes, comments,
    publish_date, city, status
)
SELECT 
    video_id,
    TRIM(channel_name),
    TRIM(title),
    TRIM(category),
    views,
    likes,
    dislikes,
    comments,
    publish_date,
    CASE 
        WHEN city = 'Bangalore' THEN 'Bengaluru'
        ELSE TRIM(city)
    END,
    CASE 
        WHEN LOWER(status) = 'active' THEN 'Active'
        WHEN LOWER(status) = 'inactive' THEN 'Inactive'
        ELSE 'Pending'
    END
FROM yt
WHERE 
    video_id IS NOT NULL
AND TRIM(channel_name) NOT IN ('', 'Unknown')
AND TRIM(title) NOT IN ('', 'Unknown');
